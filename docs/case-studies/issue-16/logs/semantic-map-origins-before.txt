#!/usr/bin/env bash
fibonacci() {
    local count=$1
    local previous=0
    local current=1
    local step
    for (( step = 0; step < count; step++ )); do
        local next=$((previous + current))
        previous=$current
        current=$next
    done
    REPLY=$previous; return
}

    fibonacci "$index"

done
